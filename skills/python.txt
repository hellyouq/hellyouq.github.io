# Классы и объекты
class Person:
    """Класс, представляющий человека"""
    
    # Атрибут класса (общий для всех экземпляров)
    species = "Homo sapiens"
    
    # Конструктор
    def __init__(self, name: str, age: int):
        self.name = name
        self.age = age
        self._private = "Приватное поле"  # protected
        self.__secret = "Секретное поле"  # private (name mangling)
    
    # Метод экземпляра
    def introduce(self) -> str:
        return f"Меня зовут {self.name}, мне {self.age} лет"
    
    # Свойство (getter)
    @property
    def is_adult(self) -> bool:
        return self.age >= 18
    
    # Setter
    @age.setter
    def age(self, value: int):
        if value < 0:
            raise ValueError("Возраст не может быть отрицательным")
        self._age = value
    
    # Class method
    @classmethod
    def create_child(cls, name: str) -> 'Person':
        return cls(name, 0)
    
    # Static method
    @staticmethod
    def is_valid_age(age: int) -> bool:
        return 0 <= age <= 150
    
    # Магические методы
    def __str__(self) -> str:
        return f"Person({self.name}, {self.age})"
    
    def __repr__(self) -> str:
        return f"Person('{self.name}', {self.age})"
    
    def __eq__(self, other) -> bool:
        if not isinstance(other, Person):
            return False
        return self.name == other.name and self.age == other.age

# Простой декоратор
def timer(func):
    import time
    
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} заняла {end - start:.2f} секунд")
        return result
    
    return wrapper

# Декоратор с параметрами
def repeat(count: int):
    def decorator(func):
        def wrapper(*args, **kwargs):
            results = []
            for _ in range(count):
                results.append(func(*args, **kwargs))
            return results
        return wrapper
    return decorator

# Использование
@timer
def slow_function():
    import time
    time.sleep(1)
    return "Готово"

@repeat(3)
def say_hello(name: str):
    return f"Hello, {name}!"

# Декоратор для логирования
def log(level: str = "INFO"):
    def decorator(func):
        def wrapper(*args, **kwargs):
            print(f"[{level}] Вызов {func.__name__} с args={args}, kwargs={kwargs}")
            result = func(*args, **kwargs)
            print(f"[{level}] {func.__name__} вернула {result}")
            return result
        return wrapper
    return decorator

@log(level="DEBUG")
def add(a, b):
    return a + b

# Декоратор для проверки типов
def type_check(expected_types):
    def decorator(func):
        def wrapper(*args, **kwargs):
            for arg, expected_type in zip(args, expected_types):
                if not isinstance(arg, expected_type):
                    raise TypeError(f"Ожидался {expected_type}, получен {type(arg)}")
            return func(*args, **kwargs)
        return wrapper
    return decorator

@type_check((int, int))
def multiply(a, b):
    return a * b

# Декораторы классов
def singleton(cls):
    instances = {}
    
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    
    return get_instance

# Итераторы
class CountDown:
    def __init__(self, start):
        self.current = start
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if self.current < 0:
            raise StopIteration
        value = self.current
        self.current -= 1
        return value

for num in CountDown(5):
    print(num)  # 5,4,3,2,1,0

# Генераторы (проще, чем итераторы)
def count_down(start):
    while start >= 0:
        yield start
        start -= 1

for num in count_down(5):
    print(num)

# Генератор бесконечной последовательности
def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

fib = fibonacci()
for _ in range(10):
    print(next(fib))

# Генераторные выражения (как list comprehension, но с круглыми скобками)
squares = (x**2 for x in range(10))

# Связывание генераторов (yield from)

# Конвейер обработки данных
def read_file(filename):
    with open(filename, 'r') as f:
        for line in f:
            yield line.strip()

def filter_lines(lines, keyword):
    for line in lines:
        if keyword in line:
            yield line

def process_lines(lines):
    for line in lines:
        yield line.upper()

# Использование пайплайна
lines = read_file('data.txt')
filtered = filter_lines(lines, 'error')
processed = process_lines(filtered)

for result in processed:
    print(result)

# Создание контекстного менеджера через класс
class DatabaseConnection:
    def __init__(self, connection_string):
        self.connection_string = connection_string
        self.connected = False
    
    def __enter__(self):
        print("Подключение к базе данных...")
        self.connected = True
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        print("Закрытие соединения...")
        self.connected = False
        if exc_type:
            print(f"Ошибка: {exc_val}")
        return False  # Не подавлять исключения
    
    def query(self, sql):
        if not self.connected:
            raise RuntimeError("Нет подключения")
        return f"Результат запроса: {sql}"

with DatabaseConnection('postgresql://localhost') as db:
    result = db.query('SELECT * FROM users')
    print(result)

# Создание контекстного менеджера через contextlib
from contextlib import contextmanager

@contextmanager
def temporary_file(filename, content):
    with open(filename, 'w') as f:
        f.write(content)
    try:
        yield filename
    finally:
        import os
        os.remove(filename)

with temporary_file('temp.txt', 'Hello') as filename:
    with open(filename, 'r') as f:
        print(f.read())

# Вложенные контекстные менеджеры

import asyncio
import aiohttp
import aiofiles

# Простая асинхронная функция
async def say_hello():
    print("Hello")
    await asyncio.sleep(1)
    print("World")

# Запуск
asyncio.run(say_hello())

# Асинхронное выполнение нескольких задач
async def main():
    task1 = asyncio.create_task(say_hello())
    task2 = asyncio.create_task(say_hello())
    await task1
    await task2

# gather - параллельное выполнение
async def main():
    results = await asyncio.gather(
        say_hello(),
        say_hello(),
        say_hello()
    )

# Асинхронные HTTP запросы
async def fetch_url(url):
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            return await response.text()

async def main():
    urls = ['https://api.example.com/1', 'https://api.example.com/2']
    results = await asyncio.gather(*[fetch_url(url) for url in urls])
    print(results)

# Асинхронная работа с файлами
async def read_async(filename):
    async with aiofiles.open(filename, 'r') as f:
        return await f.read()

# Асинхронные очереди
async def producer(queue):
    for i in range(10):
        await queue.put(i)
        await asyncio.sleep(0.1)

async def consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"Получено: {item}")
        queue.task_done()

async def main():
    queue = asyncio.Queue()
    producer_task = asyncio.create_task(producer(queue))
    consumer_task = asyncio.create_task(consumer(queue))
    await producer_task
    await queue.join()
    await queue.put(None)  # Сигнал остановки
    await consumer_task

class Category(models.Model):

    slug = models.SlugField(max_length=100, unique=True)
    description = models.TextField(blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        verbose_name_plural = "Categories"
        ordering = ['name']
    
    def __str__(self):
        return self.name
    
    def get_absolute_url(self):
        return reverse('blog:category', args=[self.slug])

class Post(models.Model):
    STATUS_CHOICES = (
        ('draft', 'Draft'),
        ('published', 'Published'),
    )
    
    title = models.CharField(max_length=200)
    slug = models.SlugField(max_length=200, unique_for_date='published_at')
    content = models.TextField()
    summary = models.TextField(max_length=500, blank=True)
    featured_image = models.ImageField(upload_to='posts/%Y/%m/', blank=True)
    
    author = models.ForeignKey(User, on_delete=models.CASCADE, related_name='posts')
    categories = models.ManyToManyField(Category, related_name='posts')
    
    status = models.CharField(max_length=10, choices=STATUS_CHOICES, default='draft')
    published_at = models.DateTimeField(default=timezone.now)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    views = models.PositiveIntegerField(default=0)
    
    class Meta:
        ordering = ['-published_at']
        indexes = [
            models.Index(fields=['published_at', 'status']),
        ]
    
    def __str__(self):
        return self.title
    
    def get_absolute_url(self):
        return reverse('blog:post_detail', args=[
            self.published_at.year,
            self.published_at.month,
            self.published_at.day,
            self.slug
        ])
    
    def increase_views(self):
        self.views += 1
        self.save(update_fields=['views'])

class Comment(models.Model):
    post = models.ForeignKey(Post, on_delete=models.CASCADE, related_name='comments')
    author = models.ForeignKey(User, on_delete=models.CASCADE, related_name='comments')
    content = models.TextField()
    created_at = models.DateTimeField(auto_now_add=True)
    is_approved = models.BooleanField(default=False)

    context_object_name = 'posts'

        queryset = Post.objects.filter(status='published')
        category_slug = self.kwargs.get('category_slug')
        if category_slug:

        if search_query:
            queryset = queryset.filter(

        child=serializers.IntegerField(),
        write_only=True,
        required=False
    )
    comment_count = serializers.IntegerField(source='comments.count', read_only=True)
    
    class Meta:
        model = Post
        fields = [
            'id', 'title', 'slug', 'content', 'summary', 
            'featured_image', 'author', 'categories', 'category_ids',
            'status', 'published_at', 'created_at', 'updated_at',
            'views', 'comment_count'
        ]
        read_only_fields = ['author', 'slug', 'views', 'created_at', 'updated_at']

        category_ids = validated_data.pop('category_ids', [])
        post = Post.objects.create(**validated_data)
        if category_ids:
            post.categories.set(category_ids)
        return post

class PostViewSet(viewsets.ModelViewSet):
    queryset = Post.objects.filter(status='published')
    serializer_class = PostSerializer
    permission_classes = [IsAuthorOrReadOnly]
    filter_backends = [DjangoFilterBackend, filters.SearchFilter, filters.OrderingFilter]
    filterset_fields = ['status', 'categories']
    search_fields = ['title', 'content']
    ordering_fields = ['published_at', 'views', 'created_at']
    ordering = ['-published_at']
    
    def get_queryset(self):
        queryset = super().get_queryset()
        if self.request.user.is_authenticated:
            if self.request.user.is_staff:
                return Post.objects.all()
            return queryset | Post.objects.filter(author=self.request.user)
        return queryset
    
    @action(detail=True, methods=['post'])
    def add_comment(self, request, pk=None):
        post = self.get_object()
        serializer = CommentSerializer(data=request.data)
        if serializer.is_valid():
            serializer.save(post=post, author=request.user)
            return Response(serializer.data, status=201)
        return Response(serializer.errors, status=400)
    
    @action(detail=True, methods=['post'])
    def toggle_like(self, request, pk=None):
        post = self.get_object()
        user = request.user
        # Логика лайков
        return Response({'status': 'ok'})

def cache_response(key: str, ttl: int = 60):
    def decorator(func):
        async def wrapper(*args, **kwargs):
            cached = redis_client.get(key)
            if cached:
                return eval(cached)
            result = await func(*args, **kwargs)
            redis_client.setex(key, ttl, str(result))
            return result
        return wrapper
    return decorator

from aiohttp import web

async def handle(request):
    name = request.match_info.get('name', "Anonymous")
    return web.Response(text=f"Hello, {name}")

async def handle_json(request):
    data = await request.json()
    return web.json_response({'received': data})

async def handle_upload(request):
    reader = await request.multipart()
    field = await reader.next()
    filename = field.filename
    size = 0
    with open(f'downloads/{filename}', 'wb') as f:
        while True:
            chunk = await field.read_chunk()
            if not chunk:
                break
            size += len(chunk)
            f.write(chunk)
    return web.json_response({'size': size, 'filename': filename})

async def websocket_handler(request):
    ws = web.WebSocketResponse()
    await ws.prepare(request)
    
    async for msg in ws:
        if msg.type == web.WSMsgType.TEXT:
            await ws.send_str(f"Echo: {msg.data}")
        elif msg.type == web.WSMsgType.BINARY:
            await ws.send_bytes(msg.data)
        elif msg.type == web.WSMsgType.ERROR:
            print(f"WebSocket error: {ws.exception()}")
    
    return ws

# Маршрутизация

app.router.add_get('/hello/{name}', handle)
app.router.add_post('/json', handle_json)
app.router.add_post('/upload', handle_upload)
app.router.add_get('/ws', websocket_handler)

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders

def send_email(to_email, subject, body, attachment_path=None):
    sender_email = "your@email.com"
    sender_password = "your_password"
    
    msg = MIMEMultipart()
    msg['From'] = sender_email
    msg['To'] = to_email
    msg['Subject'] = subject
    
    msg.attach(MIMEText(body, 'plain', 'utf-8'))
    
    if attachment_path:
        with open(attachment_path, 'rb') as attachment:
            part = MIMEBase('application', 'octet-stream')
            part.set_payload(attachment.read())
            encoders.encode_base64(part)
            part.add_header(
                'Content-Disposition',
                f'attachment; filename={attachment_path.split("/")[-1]}'
            )
            msg.attach(part)
    
    try:
        server = smtplib.SMTP('smtp.gmail.com', 587)
        server.starttls()
        server.login(sender_email, sender_password)
        server.send_message(msg)
        server.quit()
        return True
    except Exception as e:
        print(f"Error: {e}")
        return False

import requests

class APIClient:
    def __init__(self, base_url, token=None):
        self.base_url = base_url
        self.token = token
        self.session = requests.Session()
        if token:
            self.session.headers.update({
                'Authorization': f'Bearer {token}'
            })
    
    def get(self, endpoint, params=None):
        response = self.session.get(
            f'{self.base_url}/{endpoint}',
            params=params
        )
        response.raise_for_status()
        return response.json()
    
    def post(self, endpoint, data=None):
        response = self.session.post(
            f'{self.base_url}/{endpoint}',
            json=data
        )
        response.raise_for_status()
        return response.json()
    
    def put(self, endpoint, data=None):
        response = self.session.put(
            f'{self.base_url}/{endpoint}',
            json=data
        )
        response.raise_for_status()
        return response.json()
    
    def delete(self, endpoint):
        response = self.session.delete(
            f'{self.base_url}/{endpoint}'
        )
        response.raise_for_status()
        return True

def send_message(queue_name, message):
    connection = pika.BlockingConnection(
        pika.ConnectionParameters('localhost')
    )
    channel = connection.channel()
    channel.queue_declare(queue=queue_name, durable=True)

def receive_messages(queue_name):
    connection = pika.BlockingConnection(
        pika.ConnectionParameters('localhost')
    )
    channel = connection.channel()
    channel.queue_declare(queue=queue_name, durable=True)
    
    def callback(ch, method, properties, body):
        print(f"Received: {body}")
        ch.basic_ack(delivery_tag=method.delivery_tag)
    
    channel.basic_consume(
        queue=queue_name,
        on_message_callback=callback
    )
    channel.start_consuming()

async def proxy(request: Request, service: str, path: str):
    if service not in services:
        return {"error": "Service not found"}, 404
    
    url = f"{services[service]}/{path}"
    body = await request.body()
    headers = dict(request.headers)
    params = dict(request.query_params)
    
    async with httpx.AsyncClient() as client:
        response = await client.request(
            method=request.method,
            url=url,
            headers=headers,
            params=params,
            content=body
        )
        
        return response.json(), response.status_code

# Middleware
@app.middleware("http")
async def metrics_middleware(request: Request, call_next):
    start_time = time.time()
    
    # Увеличиваем счётчик соединений
    active_connections.inc()
    
    # Размер запроса
    request_size.labels(method=request.method).observe(
        len(await request.body()) if request.method in ['POST', 'PUT'] else 0
    )
    
    response = await call_next(request)
    
    # Время выполнения
    duration = time.time() - start_time
    http_request_duration.labels(
        method=request.method,
        endpoint=request.url.path
    ).observe(duration)
    
    # Счетчик запросов
    http_requests_total.labels(
        method=request.method,
        endpoint=request.url.path,
        status=response.status_code
    ).inc()
    
    # Уменьшаем счётчик соединений
    active_connections.dec()
    
    return response

ЗАКЛЮЧЕНИЕ
═══════════════════════════════════════════════════════════════════════════════

Python — один из самых универсальных и востребованных языков программирования
в современном мире. От веб-разработки до искусственного интеллекта, от 
автоматизации до научных вычислений — Python везде находит применение.

Это руководство охватывает все ключевые аспекты Python-разработки:
• Основы языка и продвинутые техники
• Веб-фреймворки (Django, FastAPI, Flask)
• Работа с базами данных
• Data Science и машинное обучение
• Автоматизация и скрипты
• Тестирование и качество кода
• DevOps и инфраструктура
• Безопасность и производительность

Ключевые принципы успешного разработчика:
1. Постоянно учитесь и экспериментируйте
2. Пишите читаемый и поддерживаемый код
3. Используйте лучшие практики и инструменты
4. Тестируйте свой код
5. Документируйте и комментируйте
6. Участвуйте в сообществе
7. Не бойтесь пробовать новое

Изучение Python — это инвестиция в вашу карьеру. Язык продолжает развиваться,
появляются новые инструменты и фреймворки. Следите за трендами, обновляйте
знания и всегда стремитесь к совершенству.

Удачи в программировании!
